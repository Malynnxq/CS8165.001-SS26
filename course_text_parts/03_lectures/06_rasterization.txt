Pseudo Code – Simple Version
void midPointLine(int x0, int y0, int x1, int y1) {
int dx = x1 - x0; // Calculate delta x
int dy = y1 - y0; // Calculate delta y
int d = 2 * dy - dx; // Initial decision variable
int dE = 2 * dy; // Increment used for move to E
int dNE = 2 * (dy - dx); // Increment used for move to NE
int x = x0, y = y0; // Start position
writePixel(x, y); // Plot the first pixel
while (x < x1) { // Loop through from x0 to x1
if (d <= 0) { // Decision is East
d += dE; x++;
} else { // Decision is North East
d += dNE; x++; y++;
}
writePixel(x, y); // Plot each computed pixel
}
}

void drawLine(int x0, int y0, int x1, int y1) {
int dx = abs(x1 - x0); // Calculate absolute delta x
int sx = (x0 < x1 ? 1 : -1); // Set step x based on direction
int dy = abs(y1 - y0); // Calculate absolute delta y
int sy = (y0 < y1 ? 1 : -1); // Set step y based on direction
int d = 2 * dy - dx; // Initial decision variable
int dE = 2 * dy; // Increment for move East or West
int dNE = 2 * (dy - dx); // Increment for move NE or SW
int x = x0, y = y0; // Start positions
writePixel(x, y); // Plot the first pixel
while (x != x1) { // Loop until target x reached
if ((d < 0) || ((d == 0) && (sx == 1))) { // Decide East or West
d += dE; x += sx;
} else { // Decide North-East or South-West
d += dNE; x += sx; y += sy;
}
writePixel(x, y); // Plot each computed pixel
}
}
